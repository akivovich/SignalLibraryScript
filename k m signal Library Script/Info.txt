Signal (Trainz)
  └─ ZmvSignalInterface          — абстрактный интерфейс светофора
      └─ ZmvSignal               — главный класс-оркестратор
          └─ KM_SignalTables     — добавляет табличные индикаторы
              └─ KM_SignalRoutePointer / KM_SignalRoutePointer2 - маршрутные указатели

Library (Trainz)
  └─ ZmvSignalLibraryInterface
      └─ ZmvSignalLibrary        — фабрика, создаёт нужный тип библиотеки

ZmvInterface                     — абстрактный интерфейс логики
  └─ ZmvBaseLibrary              — ядро логики (1676 строк)
      ├─ ZmvWRLibrary            — маневровый (Б/К)
      │   └─ ZmvWRWLibrary       — вариант Б/К/Б
      ├─ ZmvGRLibrary            — зелёный/красный
      │   └─ ZmvYGRLibrary       — жёлтый/зелёный/красный (основной)
      │       └─ ZmvWYGYRLibrary — полуавтоматы
      │           ├─ ZmvYWYRLibrary
      │           ├─ ZmvYGYRLibrary
      │           └─ ZmvWYGRLibrary
      └─ ZmvOPLibrary            — ОП - красный
          └─ ZmvRWLibrary        — красный/ПС

Команды для сценариев:
PostMessage(lightObject, "CTRL", "UpdateState", 0);    //обновить состояние					
PostMessage(lightObject, "CTRL", "MayOpen^true", 0);   //открыть светофор
PostMessage(lightObject, "CTRL", "MayOpen^false", 0);  //закрыть светофор
PostMessage(lightObject, "CTRL", "SetPS^true", 0);     //включить пригласительный сигнал
PostMessage(lightObject, "CTRL", "SetPS^false", 0);    //выключить пригласительный сигнал
